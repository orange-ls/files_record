#### 计算产品备货申请

1、页面上的两个时间选择框不再用来搜索数据，而是将他们的值作为刷新sql的where条件

sql修改为：

```sql
WITH cccs AS(
	SELECT *
	FROM crm_compute_contract_stock
	WHERE TO_DATE(SUBSTRING(stock_code FROM '\d{8}'), 'YYYYMMDD') BETWEEN {start_time} AND {end_time}
)

SELECT
		product_id,
		remark,
		crm_no,
		proj_name,
		sales,
		quot_no,
		material_code,
		material_desc,
		service_desc,
		CASE
				WHEN service_desc LIKE '%白金%' OR service_desc LIKE '%白金+%' OR service_desc LIKE '7*24*2H' THEN '7*24*2H'
				WHEN service_desc LIKE '%金牌%' OR service_desc LIKE '%金牌+%' OR service_desc LIKE '7*24*4H' THEN '7*24*4H'
				WHEN service_desc LIKE '%标准%' OR service_desc LIKE '%标准+%' OR service_desc LIKE '7*24*ND' THEN '7*24*ND'
				ELSE '基础保修'
		END AS service_time,
		'待定' AS del_address,
		'待定' AS stock_address,
		from_info,
		bundling_number,
		"number",
		apply_time
FROM (
		SELECT
				xt.product_id AS product_id,
				ck.stock_code AS remark,
				ck.crm_project_num AS crm_no,
				xq.crm_name AS proj_name,
				xq.pre_sales_name || '-' || xq.pre_sales_it_code AS sales,
				xq.quot_no,
				xt.transfer_sap_no AS material_code,
				xt.transfer_spec AS material_desc,
				CASE
						WHEN xt.transfer_spec LIKE '%维保服务%' AND xt.transfer_sap_no LIKE '80-%' THEN xt.transfer_spec
						WHEN xt.transfer_spec LIKE '%介质保留%' AND xt.transfer_sap_no LIKE '80-%' THEN '+介质保留'
						ELSE '无鲲泰服务器维保服务'
				END AS service_desc,
				'PO通知单' AS from_info,
				COALESCE(bn.bundling_number, xt.transfer_sap_no) AS bundling_number,
				xt.transfer_quantity * xp.pro_numb AS "number",
				ck.apply_time as apply_time
		FROM
				cccs ck
				LEFT JOIN xc_quotation xq ON ck.crm_project_num = xq.crm_no
				LEFT JOIN xc_product xp ON xp.quot_no = xq.quot_no
				LEFT JOIN xc_product_detail xd ON xd.pro_id = xp."id"
				LEFT JOIN xc_product_detail_transfer xt ON xt.product_detail_id = xd."id"
				LEFT JOIN bundling_part_number bn ON xt.transfer_sap_no = bn.material_mode
		WHERE
-- 				plan_next_deal = '继续流转' AND
				xt.transfer_sap_no IS NOT NULL
				AND xt.transfer_sap_no NOT IN (SELECT material_mode FROM non_electronic_materials)
				AND xd.mat_type != '主机型号'
				AND xp.del_pro_flag = '0'
) AS service_desc_table
ORDER BY
		remark,product_id
```
